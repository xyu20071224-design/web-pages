1. 一维数组：分配与访问原第 3–6 页

数组的基本原则只有一条：T A[L] 是「类型为 T、长度为 L 的数组」，在内存里占一段连续的 L × sizeof(T) 字节。没有长度信息跟着数组走、也没有任何边界检查 —— 这两点后面都会反复回来。

课件给了四个声明，起始地址都记作 x。注意最后一种：char *p[3] 是「3 个指针组成的数组」，每个指针在 x86-64 上占 8 字节：

交互：选一个声明，看它的字节地图（地址现算）
从这张图里读出三件事
① 数组占的总字节数 = 元素个数 × 单个元素的字节数，一格不多一格不少；② 第 i 个元素的地址 = x + i × sizeof(T) —— 「乘以元素大小」这一步，在机器级就是寻址操作数里的缩放因子（后面的 (%rdi,%rsi,4)）；③ char *p[3] 与 double a[3] 占地一样大 —— 因为指针和 double 都是 8 字节。

2.2 get_digit：一条指令完成下标算术原第 8 页

int get_digit(zip_dig z, int digit)
{
  return z[digit];
}
# %rdi = z（数组起始地址）  %rsi = digit
  movl (%rdi,%rsi,4), %eax  # z[digit]

上面的模拟器算的就是这条指令：内存操作数 (%rdi,%rsi,4) = 基址 %rdi + 下标 %rsi × 缩放 4。数组大小编译期已知 → 缩放因子直接写进指令，连一次乘法都不用。（课件在这页汇编旁标了 IA32 —— 32 位形式的写法，寻址方式与 x86-64 完全一致。）

2.3 zincr：循环给每个元素加一原第 9 页

void zincr(zip_dig z) {
  size_t i;
  for (i = 0; i < ZLEN; i++)
    z[i]++;
}
# %rdi = z
  movl $0, %eax            #  i = 0
  jmp .L3                 #  goto middle
.L4:                     #  loop:
  addl $1, (%rdi,%rax,4)  #  z[i]++
  addq $1, %rax           #  i++
.L3:                     #  middle:
  cmpq $4, %rax           #  i : 4
  jbe .L4                 #  if <=, goto loop
  rep; ret
初始化
movl $0, %eax → i = 0；jmp 直接跳到中间的判断
do-while 形态：先跳过循环体，先判断再进循环（第 6 讲的「guarded」模式）。
循环体：z[i]++
addl $1, (%rdi,%rax,4)
对内存里的那个 int 直接 +1 —— 读改写一条指令完成，缩放因子 4 还是元素大小。
更新与判断
addq $1, %rax；cmpq $4, %rax；jbe .L4
i 是 size_t（无符号），所以判断用 jbe（无符号 ≤）。i:4 是 AT&T 风格「比较 i 与 4」。
收尾
rep; ret
rep 前缀放在 ret 前只为避免 AMD 分支预测器的 penalty —— 它不改变 ret 的语义。

3. 嵌套（多维）数组原第 10–15 页

T A[R][C] 是「R 个元素、每个元素又是 C 个 T 的数组」。整个二维数组仍然是一整块连续内存，按行主序（row-major）铺开：先铺第 0 行，再铺第 1 行……

3.1 尺寸与地址公式原第 10、14 页

量公式int 数组（K = 4）
总大小R × C × K4·R·C 字节（课件原话）
第 i 行的起始地址A + i × (C × K)A + 4·C·i
元素 A[i][j] 的地址A + i × (C × K) + j × K = A + (i × C + j) × KA + (i·C + j)·4
一维公式是二维公式的特例
把「一行 C 个元素」看成一个「大元素」，二维数组就是「R 个大元素的一维数组」—— 行地址公式和 1.1 节的 x + i·sizeof(T) 一模一样，只是这里的 sizeof(T) 换成了 C × K。列下标 j 再在行内做一次一维寻址。
交互：二维数组地址计算器 —— 改维度、改下标，网格与地址全部现算

3.2 行访问：pgh[index] 是个「小数组」原第 12–13 页

#define PCOUNT 4
zip_dig pgh[PCOUNT] =
  {{1, 5, 2, 0, 6},
   {1, 5, 2, 1, 3},
   {1, 5, 2, 1, 7},
   {1, 5, 2, 2, 1}};  /*「zip_dig pgh[4]」等价于「int pgh[4][5]」*/

课件原图给了五个边界地址：76、96、116、136 是四行的起始地址（每行 20 字节），156 是整个数组的末尾（恰好在最后一格之后）。行访问代码把「行地址 = pgh + 20·index」拆成了两条 leaq：

int *get_pgh_zip(int index)
{
  return pgh[index];
}
# %rdi = index
  leaq (%rdi,%rdi,4),%rax  #  5 * index
  leaq pgh(,%rax,4),%rax  #  pgh + (20 * index)
第 1 步：5 × index
leaq (%rdi,%rdi,4),%rax → %rax = index + 4·index = 5·index
leaq 的第二操作数是「基址 + 缩放」的地址算术 —— 这里拿来当乘法器用：C=5 拆成 1+4。
第 2 步：×4 再加基址
leaq pgh(,%rax,4),%rax → %rax = pgh + 4·(5·index) = pgh + 20·index
缩放因子 4 = sizeof(int)。两次 leaq 合起来就是行地址公式 A + i·(C·K)。
返回
%rax 里装的是地址（int *）
返回「第 index 行」这个长度为 5 的 int 数组的首地址 —— 数组名退化为指针。

3.3 元素访问：一条乘法都不用原第 15 页

int get_pgh_digit(int index, int dig)
{
  return pgh[index][dig];
}
  leaq (%rdi,%rdi,4), %rax  #  5*index
  addl %rax, %rsi          #  5*index + dig
  movl pgh(,%rsi,4), %eax  #  M[pgh + 4*(5*index + dig)]

把 (i·C + j)·K 改写成 (5i + j)·4：先算 5i（leaq），加上 j（addl），最后一次性 ×4 + 基址（movl 的缩放寻址）。C·K 能拆成「缩放因子 + 小整数乘法」时，编译器全程不用 imul。

边界提醒：课件第 11 页的「pgh[4];」标签
课件内存图旁写着 zip_dig pgh[4]; —— 它是整张图的声明标注（等价于 int pgh[4][5]），不是「还有一个 pgh[4] 元素」：图里的 5 个地址是 4 行的起点（76/96/116/136）加数组末尾（156）。数组的合法行下标只有 0–3。

4. 多级数组：指针的数组原第 16–18 页

把「数组的数组」换成「指针的数组」，同样的 univ[index][digit] 写法，机器级的地址计算完全变样 —— 这次要读两次内存。

4.1 例子的内存图原第 16 页

zip_dig cmu = { 1, 5, 2, 1, 3 };  /* @16 */
zip_dig mit = { 0, 2, 1, 3, 9 };  /* @36 */
zip_dig ucb = { 9, 4, 7, 2, 0 };  /* @56 */

4.2 两次访存的地址计算原第 17 页

int get_univ_digit(size_t index, size_t digit)
{
  return univ[index][digit];
}
  salq $2, %rsi             #  4*digit
  addq univ(,%rdi,8), %rsi  #  p = univ[index] + 4*digit
  movl (%rsi), %eax        #  return *p
  ret

计算的是 Mem[Mem[univ + 8·index] + 4·digit]：第一次访存从 univ 数组里取出第 index 个指针（所以缩放是 8），第二次访存才从这个指针指向的数组里取元素（缩放 4）。行与行之间不保证连续 —— 每行的位置都存在指针里。

5. N × N 矩阵：固定尺寸 vs 变长尺寸原第 19–21 页

矩阵就是二维数组。但当行数 n 要到运行时才知道时，地址公式里的 n·i 就不能再拆成 leaq 的常数戏法 —— 得老老实实做一次整数乘法。

5.1 三种写法原第 19 页

#define N 16
typedef int fix_matrix[N][N];
/* 固定维度：编译期就知道 N */
int fix_ele(fix_matrix a, size_t i, size_t j)
{  return a[i][j];  }
#define IDX(n, i, j) ((i)*(n)+(j))
/* 变长维度 · 显式下标：传统做法，手动摊平成一维 */
int vec_ele(size_t n, int *a, size_t i, size_t j)
{  return a[IDX(n,i,j)];  }
/* 变长维度 · 隐式下标：C99 的变长数组，gcc 支持 */
int var_ele(size_t n, int a[n][n], size_t i, size_t j)
{  return a[i][j];  }

5.2 固定尺寸：C·K = 64 全程常数原第 20 页

# a in %rdi, i in %rsi, j in %rdx
  salq $6, %rsi            #  64*i
  addq %rsi, %rdi          #  a + 64*i
  movl (%rdi,%rdx,4), %eax  #  M[a + 64*i + 4*j]

C = 16、K = 4，行宽 64 字节是常数 → 64·i 就是左移 6 位（salq $6），列下标交给 movl 的缩放寻址。没有任何乘法指令。

5.3 变长尺寸：一次 imul 逃不掉原第 21 页

# n in %rdi, a in %rsi, i in %rdx, j in %rcx
  imulq %rdx, %rdi        #  n*i
  leaq (%rsi,%rdi,4), %rax  #  a + 4*n*i
  movl (%rax,%rcx,4), %eax  #  a + 4*n*i + 4*j

行宽 4n 不再是常数，只能 imulq 算出 n·i，再用 leaq 的缩放 4 乘掉 sizeof(int)。注意参数顺序变了：变长数组把 n 排在第一个参数（%rdi），a 退到 %rsi。

交互：fix_ele vs var_ele —— 填 n、i、j，看每条指令后的寄存器
两种模式的成本差在哪
fix：移位 + 加 + 读（3 条）；var：乘 + leaq + 读（3 条）。差别不在条数，而在 imul 本身 —— 它比移位慢。所以 vec_ele 的 IDX 宏写法（把 n 显式传进来）和 var_ele 的 C99 变长数组，生成的地址算术本质相同；只有固定 N 时编译器才能把乘法整个消掉。

pgh[0][0] 是标准二维写法；linear_zip[7] 把整个二维数组摊平后取第 7 个 int（= pgh[1][2]，因为 7 = 1·5 + 2）；*(linear_zip + 8) 是同一个摊平操作的指针写法（= pgh[1][3]）；zip2[1] 先用 pgh[2] 拿到第 2 行的行地址、再在行内取下标 1。课件打印的 result: 9 与页面现算一致 —— 你可以改 k/m/t 再算，摊平映射永远是 linear_zip[k] = pgh[k/5][k%5]（合法范围 k = 0…19，与 3.1 节的地址公式 A + (i·C + j)·K 互相印证）。
交互：两道判断题

7. 结构体：布局与访问原第 25–27 页

结构体 = 一块足够装下所有字段、字段按声明顺序排列的内存。哪怕换个顺序能省空间，编译器也不重排 —— 字段偏移完全由编译期决定，机器级程序对「结构体」三个字一无所知，只认识偏移。

7.1 struct rec：一张 32 字节的地图原第 25 页

struct rec {
  int a[4];         /* 偏移 0–15，16 字节 */
  size_t i;         /* 偏移 16–23，8 字节 */
  struct rec *next; /* 偏移 24–31，8 字节 */
};                  /* 共 32 字节 */
交互：struct rec 字节地图 —— 点字段名看覆盖范围；填 idx 算 &r->a[idx]

7.2 取成员指针：偏移在编译期就定了原第 26 页

int *get_ap(struct rec *r, size_t idx)
{
  return &r->a[idx];
}
# r in %rdi, idx in %rsi
  leaq (%rdi,%rsi,4), %rax  #  r + 4*idx
  ret

&r->a[idx] = r + 4·idx：数组的基址偏移（0）不用加，缩放 4 来自 int。假如取的是 &r->next，那就是 r + 24 —— 偏移 24 是编译期常量，直接写进指令。

7.3 链表遍历：下标本身存在内存里原第 27 页

void set_val(struct rec *r, int val)
{
  while (r) {
    int i = r->i;  /* Element i */
    r->a[i] = val;
    r = r->next;
  }
}
.L11:                     #  loop:
  movslq 16(%rdi), %rax   #   i = M[r+16]
  movl %esi, (%rdi,%rax,4)#   M[r+4*i] = val
  movq 24(%rdi), %rdi     #   r = M[r+24]
  testq %rdi, %rdi        #   Test r
  jne .L11                #   if !=0 goto loop

注意第三个字段的偏移：next 在 24。这一页的声明把 i 写成 int（第 25 页是 size_t）—— 两种写法的字段偏移相同（int 之后垫 4 字节、size_t 之后就不用垫，next 都落在 24），但「要写内存的字段，它的下标 i 也是从内存里读出来的」（movslq 16(%rdi)）这件事不变：寻址模式 (%rdi,%rax,4) 的「下标寄存器」内容来自上一次访存。

交互：set_val 步进器 —— 看寄存器与内存怎么跟着 while 循环走
movslq 是干嘛的
movslq 16(%rdi), %rax = 「load 32 位有符号数、符号扩展到 64 位」。r->i 声明为 int（4 字节），而寻址要用 64 位的 %rax —— 所以取出来的 32 位值要扩展。若按第 25 页把 i 声明成 size_t（8 字节），这条就变成普通的 movq。页面下方的步进器按课件汇编原文（int 版）演示。

8. 对齐：为什么结构体会「虚胖」原第 28–35 页

对齐规则两句话：基本类型占 K 字节，它的地址必须是 K 的倍数；结构体整体的对齐要求 K 取「所有字段里最大的 K」，起始地址与总长度都必须是 K 的倍数。中间垫的叫内部填充，末尾垫的叫外部填充。

8.1 x86-64 的对齐要求原第 30 页

K类型地址要求（课件原文）
1char …地址无限制
2short …地址最低 1 位必须是 02（2 的倍数）
4int、float …地址最低 2 位必须是 002（4 的倍数）
8double、long、char * …地址最低 3 位必须是 0002（8 的倍数）
16long double（Linux 上的 GCC）地址最低 4 位必须是 00002（16 的倍数）
为什么要对齐
课件第 29 页给了三条动机：内存按 4/8 字节的对齐块访问（具体依赖系统）；跨「四字边界」的数据读写效率低；数据若横跨两个虚拟内存页，访存会更麻烦。编译器的对策：往结构体里插空隙（gaps），保证每个字段都对齐。

8.2 课件的三个结构体原第 28、31–34 页

struct S1 {        struct S2 {        struct S3 {
  char c;            double v;          short i;
  int i[2];          int i[2];          float v;
  double v;          char c;            short j;
} *p;               } *p;               } a[10];
结构体偏移（课件标注）填充大小K
S1c=0, i[0]=4, i[1]=8, v=16内部：1–3（3 字节）+ 12–15（4 字节）；外部：0248
S2v=0, i[0]=8, i[1]=12, c=16内部：0；外部：17–23（7 字节）248
S3i=0, v=4, j=8内部：2–3 与 10–11；每个元素外部：0124
同一批字段，两种命运
S1 与 S2 字段完全相同：S2 把最大的字段放开头，内部填充归零；但两者的总大小都得是 K=8 的倍数，于是 S1 靠内部填充、S2 靠末尾 7 字节外部填充，殊途同归都是 24 字节。S3 则演示了「结构体数组」：a[10] 的每个元素都是 12 字节（K=4 的倍数），a[idx].j 的地址 = a + 12·idx + 8 —— 汇编里写成 movzwl a+8(,%rax,4)，其中 %rax = 3·idx（课件第 34 页）。

8.3 省空间：把大的放前面原第 35 页

声明布局大小
struct S4 { char c; int i; char d; };c=0, 垫3, i=4–7, d=8, 垫312 字节
struct S5 { int i; char c; char d; };i=0–3, c=4, d=5, 垫28 字节

9. 浮点：数据搬进 XMM 寄存器原第 37–42 页

浮点数不走通用寄存器：参数放在 %xmm0、%xmm1…，返回值放 %xmm0，而且全部 XMM 寄存器都是 caller-saved（调用者保存）。这套规则属于 SSE/AVX 家族 —— 课件第 37 页的历史线：x87（祖传、非常丑）→ SSE（服务器机支持，向量指令的特殊用法）→ AVX（最新，与 SSE 相似，书里讲的就是它）。

9.1 一个 XMM 寄存器的七种读法原第 38 页

XMM 寄存器共 16 个，每个 16 字节。同一份 16 字节可以按不同「宽度 × 个数」来解释：

解释方式份数每份宽度
8 位整数161 B
16 位整数82 B
32 位整数44 B
单精度浮点 float44 B
双精度浮点 double28 B
标量单精度（只用低 4 字节）14 B
标量双精度（只用低 8 字节）18 B

9.2 标量 vs SIMD：一条指令算几个原第 39 页

指令含义计算几个
addss %xmm0,%xmm1标量单精度（scalar single）只算最低 1 个 float
addps %xmm0,%xmm1打包单精度（packed，后缀没了 s）同时算 4 个 float
addsd %xmm0,%xmm1标量双精度（scalar double）只算最低 1 个 double
addpd %xmm0,%xmm1打包双精度同时算 2 个 double

9.3 传参、访存与函数原第 40–41 页

float fadd(float x, float y)    double dadd(double x, double y)
{  return x + y;                {  return x + y;
}                                }
# x in %xmm0, y in %xmm1
  addss %xmm1, %xmm0             addsd %xmm1, %xmm0
  ret                            ret

整数（和指针）参数照旧走通用寄存器；浮点参数走 XMM。于是一个函数可以同时收两种参数（课件第 41 页 dincr：指针 p 在 %rdi、浮点 v 在 %xmm0）：

double dincr(double *p, double v)
{
  double x = *p;
  *p = x + v;
  return x;
}
# p in %rdi, v in %xmm0
  movapd %xmm0, %xmm1  #  Copy v
  movsd (%rdi), %xmm0  #  x = *p
  addsd %xmm0, %xmm1   #  t = x + v
  movsd %xmm1, (%rdi)  #  *p = t
  ret
XMM 与内存之间也分「标量 / 打包」
movsd 一次搬 8 字节（一个 double）；movapd 在 XMM 寄存器之间整份搬 16 字节（aligned packed double）。后缀的 s/p 规则和运算指令一致 —— 一堆指令长得像，是课件第 42 页抱怨「Lots of instructions」的原因。

9.4 浮点比较：ucomiss / ucomisd原第 42 页

浮点比较用 ucomiss（单精度）/ ucomisd（双精度），设置条件码 ZF、PF、CF（并把 OF、SF 清零）。四个结果对应的位组合 —— 无序（UNORDERED）占了一整行，因为 NaN 和任何数（包括它自己）比较都是「无序」：

关系ZF PF CF
UNORDERED（有 NaN）1 1 1
GREATER_THAN0 0 0
LESS_THAN0 0 1
EQUAL1 0 0

顺带一提：课件同页给了一条把 XMM 清零的惯用法 —— xorpd %xmm0, %xmm0（自己异或自己）；其他常数则从内存加载。

10.1 五个声明，一个 deref 规则原第 47、49 页

剥一个 * 的规则：对指针剥掉一层指向；对数组剥掉最外层维度（数组名此时退化为指针）。剥不动（剩下的不是指针也不是数组）就编译不过：

声明读法本体是什么
int A1[3]3 个 int 的数组已分配的数组（12 B）
int *A2[3]3 个 int* 的数组已分配的指针数组（24 B），指针本身未初始化
int (*A3)[3]指向「3 个 int 的数组」的指针一个未分配的指针（8 B）
int (*A4[3])同 A2 —— 括号不改变它已分配的指针数组（24 B）
int (*A5[3])[5]3 个指针，各指向「5 个 int 的数组」已分配的指针数组（24 B）
交互：声明大表自测 —— Size 列自己填，判分基准 = 真实编译器的 sizeof
这张表的三个基准各有各的来源
Size 列由页面按 deref 规则现算，已与 clang 在 x86-64 下对全部 5 个声明 × 4 层 deref 的 sizeof 输出逐项比对（0 项不一致）；Cmp 列也与 clang 的真实编译结果核对过；Bad 列沿用课件第 46/48/51 页自己的答案 —— 判断标准是「这个表达式是否可能解引用了未分配的对象」。
Bad 列的口径
「已分配 / 未分配」是课件图的用词：A1 的格子是分配了的 int；A2/A4/A5 数组本体已分配，但里面的指针值未初始化，解引用它们就是坏引用；A3 本身就是一根没分配的指针。所以 **A2（只读到指针值为止）Bad = N，而 ***A2（解引用那根野指针）Bad = Y。这些表达式若真执行，行为未定义 —— 表里的 Y/N 是「可能的坏引用」。
